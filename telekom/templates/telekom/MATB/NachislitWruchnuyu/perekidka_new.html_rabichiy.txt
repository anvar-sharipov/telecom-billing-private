function priceAlemInputed(event) {
        // Получаем текущее значение поля ввода
        const input = event.target;
        let value = input.value;

        // Заменяем запятую на точку
        value = value.replace(/,/g, '.');

        // Регулярное выражение, которое разрешает цифры, точку и минус
        // Минус допускается только в начале строки
        const filteredValue = value.replace(/(?!^[-]?$)[^0-9.-]/g, '') // Убираем недопустимые символы
                                   .replace(/(?!^[-])[-]+/g, '') // Убираем лишние минусы
                                   .replace(/(\..*?)\..+/g, '$1'); // Убираем лишние точки

        // Проверяем, не начинается ли строка с точки
        if (filteredValue.startsWith('.')) {
            input.value = '0' + filteredValue; // Добавляем 0 перед точкой
        } else {
            input.value = filteredValue; // Устанавливаем отфильтрованное значение обратно в поле ввода
        }

        // Проверяем корректный ввод отрицательного числа и дополнительную точку.
        // Убедимся в том, что несмотря на наличие символов, и правильный формат перед точкой не нарушается.
        let segments = input.value.split('.');

function priceAlemInputed(event) {
        // Получаем текущее значение поля ввода
        const input = event.target;
        let value = input.value;

        // Заменяем запятую на точку
        value = value.replace(/,/g, '.');

        // Регулярное выражение, которое разрешает цифры, точку и минус
        // Минус допускается только в начале строки
        const filteredValue = value.replace(/(?!^[-]?$)[^0-9.-]/g, '') // Убираем недопустимые символы
                                   .replace(/(?!^[-])[-]+/g, '') // Убираем лишние минусы
                                   .replace(/(\..*?)\..+/g, '$1'); // Убираем лишние точки

        // Проверяем, не начинается ли строка с точки
        if (filteredValue.startsWith('.')) {
            input.value = '0' + filteredValue; // Добавляем 0 перед точкой
        } else {
            input.value = filteredValue; // Устанавливаем отфильтрованное значение обратно в поле ввода
        }

        // Проверяем корректный ввод отрицательного числа и дополнительную точку.
        // Убедимся в том, что несмотря на наличие символов, и правильный формат перед точкой не нарушается.
        let segments = input.value.split('.');
        if (segments.length > 2) {
            // Если есть больше одной точки, удаляем все после первой.
            input.value = segments[0] + '.' + segments[1];
        }
    }
